# ── Banco de Dados (Postgres do próprio compose) ──────────
# Dirige o contêiner do Postgres E a conexão da API. A senha pode ter
# qualquer caractere, inclusive espaço e @ / : ? # — a API monta a conexão
# com percent-encoding. Não use aspas em volta.
#
# ATENÇÃO: DB_PASSWORD só tem efeito na PRIMEIRA inicialização. Se o volume
# já existe de um deploy anterior, trocar aqui NÃO muda a senha do banco —
# o log vai acusar "password authentication failed". Nesse caso, apague o
# volume (se não houver dados) ou rode no contêiner do Postgres:
#   ALTER USER cupons WITH PASSWORD 'a-senha-nova';
DB_NAME=cupons
DB_USER=cupons
DB_PASSWORD=SUBSTITUIR_openssl_rand_base64_24

# ── Cloudflare Tunnel ─────────────────────────────────────
# Em dash.cloudflare.com -> Zero Trust -> Networks -> Tunnels:
#   1. Create a tunnel (tipo Cloudflared), dê o nome fb-cupons
#   2. Copie o token que aparece no comando de instalação
#   3. Em Public Hostname, adicione:
#        Subdomain: cupom      Domain: fbtax.cloud
#        Service:   HTTP       URL:    frontend:80
#   O túnel cria o registro DNS sozinho. Se já existir um registro A manual
#   para "cupom", apague — os dois disputariam o mesmo nome.
CLOUDFLARE_TUNNEL_TOKEN=SUBSTITUIR_TOKEN_DO_TUNEL

# ── Sessão — gere com: openssl rand -hex 64 ───────────────
# Mínimo de 32 caracteres; a API recusa subir com menos.
JWT_SECRET=SUBSTITUIR_openssl_rand_hex_64
SESSAO_HORAS=12

# ── Domínio ───────────────────────────────────────────────
DOMINIO_URL=https://cupom.fbtax.cloud

# ── Primeiro acesso ───────────────────────────────────────
# Criados UMA ÚNICA VEZ, na primeira subida com a tabela de usuários vazia.
# Trocar depois não altera o usuário existente.
#
# Se ADMIN_SENHA ficar vazia ou tiver menos de 8 caracteres, a API NÃO
# falha: sorteia uma senha forte e a anuncia no log, no aviso
# "administrador criado com SENHA SORTEADA".
ADMIN_EMAIL=claudiosousadebezerra@gmail.com

# ── Leitura dos cupons via Claude (visão) ─────────────────
# OPCIONAL. Sem a chave o sistema sobe e funciona, menos a aba Importar:
# cadastros, revisão manual, cobrança e Excel seguem valendo.
# A chave fica só no servidor — nunca vai ao frontend nem ao log.
ANTHROPIC_API_KEY=

# Endereço da API. Vazio = api.anthropic.com.
#
# Preencha SÓ se a chave for de um gateway compatível com a API Messages da
# Anthropic (OmniRoute, proxy corporativo). Uma chave de gateway enviada para
# api.anthropic.com é recusada com 401 "invalid x-api-key" — o endereço tem de
# acompanhar a chave. O log de subida mostra para onde as requisições vão, no
# campo "endereco" da linha "leitura via Claude habilitada".
#
# Atenção: o gateway precisa expor /v1/messages no formato da Anthropic. Se
# ele só fala o formato da OpenAI, não serve — o SDK usado aqui é o da
# Anthropic.

# Modelo da leitura normal e do botão "reler com modelo mais forte".
CLAUDE_MODEL=claude-opus-5

# Esforço de raciocínio: low, medium, high, xhigh, max. Cupom apagado
# melhora com esforço maior, e cada degrau custa mais.
CLAUDE_ESFORCO=high
CLAUDE_TIMEOUT_S=180

# Páginas lidas em paralelo. Acima de 3 cresce o risco de 429 na API.

# Lado maior da imagem enviada à API. 1568 é o ponto em que a própria API
# pararia de ganhar resolução. Use 0 para enviar sem redimensionar.

# Teto de leituras por dia, para controle de custo. 0 desliga o limite.
